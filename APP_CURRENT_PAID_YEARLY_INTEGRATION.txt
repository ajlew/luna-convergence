@st.cache_data(show_spinner=False, ttl=86400)
def _cached_paid_yearly_editorial(
    packet_json: str,
    natal_profile_value: str,
    timezone_name: str,
    main_focus: str,
    personal_question: str,
    base_url: str,
    model: str,
    _api_key: str,
) -> dict:
    """One cached Paid Yearly call. Provider failures are cached too."""
    snapshot = _snapshot_from_encoded_natal_profile(
        natal_profile_value,
        timezone_name=timezone_name,
    )
    if snapshot is None:
        return {
            "editorial": None,
            "error": "The paid Yearly natal geometry could not be reconstructed.",
        }

    try:
        editorial = generate_paid_yearly_editorial(
            json.loads(packet_json),
            snapshot=snapshot,
            main_focus=main_focus,
            personal_question=personal_question,
            base_url=base_url,
            model=model,
            api_key=_api_key,
        )
    except Exception as exc:
        message = " ".join(str(exc or "Paid Yearly voice unavailable").split())[:700]
        if _api_key:
            message = message.replace(_api_key, "[redacted]")
        return {"editorial": None, "error": message}

    return {"editorial": editorial, "error": ""}


def _build_paid_yearly_product(
    *,
    natal_profile_value: str,
    start_date: date,
    end_date: date,
    timezone_name: str,
    main_focus: str,
    personal_question: str,
) -> dict:
    """Build the finished paid Yearly product from deterministic astrology + one Luna article."""
    snapshot = _snapshot_from_encoded_natal_profile(
        natal_profile_value,
        timezone_name=timezone_name,
    )
    if snapshot is None:
        raise ValueError(
            "The paid Year Ahead is missing the derived natal geometry required for personal transits."
        )

    timing_report = build_timing_map(
        snapshot,
        start_date=start_date,
        timezone_name=timezone_name,
        max_stories=10,
    )
    if timing_report.end_date != end_date:
        raise ValueError(
            "The paid Year Ahead period does not match the required rolling 365-day window."
        )

    packet = build_year_packet(snapshot, timing_report)

    if not _luna_voice_ready():
        raise RuntimeError(
            "Luna Voice is not configured, so the paid Yearly report has not been published."
        )

    cached = _cached_paid_yearly_editorial(
        json.dumps(packet.to_dict(), ensure_ascii=False, sort_keys=True, default=str),
        natal_profile_value,
        timezone_name,
        str(main_focus or "General overview"),
        str(personal_question or ""),
        LUNA_VOICE_BASE_URL,
        LUNA_VOICE_MODEL,
        LUNA_VOICE_API_KEY,
    )
    editorial = cached.get("editorial")
    editorial_error = str(cached.get("error") or "")

    expected_issues = len(packet.games)
    actual_issues = len(getattr(editorial, "issues", ()) or ()) if editorial is not None else 0
    word_count = int(getattr(editorial, "word_count", 0) or 0) if editorial is not None else 0

    # Paid Monthly does not knowingly publish a broken story. Paid Yearly follows
    # the same principle: no tiny dashboard fallback is delivered as the paid product.
    if (
        editorial is None
        or not bool(getattr(editorial, "voice_complete", False))
        or actual_issues != expected_issues
        or word_count < 2600
    ):
        detail = editorial_error or (
            "Luna did not return the complete long-form annual reading "
            f"({actual_issues}/{expected_issues} issue chapters, {word_count} words)."
        )
        raise RuntimeError(detail)

    return {
        "snapshot": snapshot,
        "timing_report": timing_report,
        "packet": packet,
        "editorial": editorial,
    }


def _render_paid_yearly_product(

    )
